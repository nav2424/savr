# RevenueCat API Keys Template
# Copy these lines to your .env file

# iOS API Key (Apple App Store)
EXPO_PUBLIC_REVENUECAT_IOS_API_KEY=your_ios_api_key_here

# Android API Key (Google Play Store)  
EXPO_PUBLIC_REVENUECAT_ANDROID_API_KEY=your_android_api_key_here

# Web/Browser API Key (for Expo Go testing)
EXPO_PUBLIC_REVENUECAT_WEB_API_KEY=your_web_api_key_here

# Test Store API Key (use this in Expo Go for native RevenueCat testing)
# EXPO_PUBLIC_RC_TEST_STORE_API_KEY=rcb_test_store_key_here

# Optional: specific offering ID (uses "Current" if unset)
# EXPO_PUBLIC_REVENUECAT_OFFERING_ID=default

# Optional: entitlement identifier (defaults to "pro")
# EXPO_PUBLIC_REVENUECAT_ENTITLEMENT_ID=Pro
